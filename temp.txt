function safeText(value) {  
  return String(value ?? "").replace(/[&<>"']/g, c => ({  
    "&":"&amp;",  
    "<":"&lt;",  
    ">":"&gt;",  
    '"':"&quot;",  
    "'":"&#39;"  
  }[c]));  
}  
  
function getApiKey() {  
  try {

      return String(window.AndroidBridge.getApiKey() || "");  
    }  
  } catch (e) {  
    console.warn("Android API bridge unavailable", e);  
  }  
  return "";  
}  
  
async function apiRequest(url) {  
  const response = await fetch(url);  
  const data = await response.json();  
  if (!response.ok || data.error) {  
    throw new Error(data.error?.message || "YouTube API request failed");  
  }  
  return data;  
}

function getPlaylistId(raw) {  
  try {  
    const u = new URL(raw.trim());  
    const list = u.searchParams.get("list");  
    if (list && /^[A-Za-z0-9_-]{10,}$/.test(list)) return list;  
    if (!u.hostname && /^[A-Za-z0-9_-]{10,}$/.test(raw.trim())) return raw.trim();  
  } catch (_) {  
    if (/^[A-Za-z0-9_-]{10,}$/.test(raw.trim())) return raw.trim();  
  }  
  return "";  
}  
  
function parsePlaylistFile(text) {  
  const seen = new Set();  
  const result = [];  
  text.split(/\r?\n/).forEach(line => {  
    const trimmed = line.trim();  
    if (!trimmed || trimmed.startsWith("#")) return;  
    const parts = trimmed.split("|");  
    const possibleTitle = parts.length > 1 ? parts[0].trim() : "";  
    const link = parts.length > 1 ? parts.slice(1).join("|").trim() : trimmed;  
    const playlistId = getPlaylistId(link);  
    if (!playlistId || seen.has(playlistId)) return;  
    seen.add(playlistId);  
    result.push({  
      playlistId,  
      title: possibleTitle || `YouTube Playlist ${result.length + 1}`,  
      thumbnail: "",  
      customTitle: !!possibleTitle  
    });  
  });  
  return result;  
}  
  
async function addYouTubeDetails(entries) {  
  const apiKey = getApiKey();  
  if (!apiKey || !entries.length) return;  
  for (let i = 0; i < entries.length; i += 50) {  
    const batch = entries.slice(i, i + 50);  
    const ids = batch.map(x => x.playlistId).join(",");  
    const url = "https://www.googleapis.com/youtube/v3/playlists?part=snippet&id=" + encodeURIComponent(ids) + "&key=" + encodeURIComponent(apiKey);  
    const data = await apiRequest(url);  
    const byId = new Map((data.items || []).map(x => [x.id, x.snippet || {}]));  
    batch.forEach(item => {  
      const info = byId.get(item.playlistId);  
      if (!info) return;  
      if (!item.customTitle) item.title = info.title || item.title;  
      item.thumbnail =  
        info.thumbnails?.maxres?.url ||  
        info.thumbnails?.standard?.url ||  
        info.thumbnails?.high?.url ||  
        info.thumbnails?.medium?.url ||  
        info.thumbnails?.default?.url || "";  
    });  
  }  
}

  const refresh = document.getElementById("refreshBtn");

    const response = await fetch(PLAYLISTS_URL + "?v=" + Date.now(), { cache: "no-store" });  
    if (!response.ok) throw new Error("Could not read playlists.txt (HTTP " + response.status + ")");  
    const text = await response.text();  
    playlistEntries = parsePlaylistFile(text);  
  
    if (!playlistEntries.length) {  
      showEmpty("Your playlist library is empty", "Add YouTube playlist links to playlists.txt in your GitHub repository.");  
      setStatus("No playlist links found.");  
      return;  
    }  
  
    try {  
      await addYouTubeDetails(playlistEntries);  
    } catch (err) {  
      console.warn(err);  
    }

    setStatus("Library updated.");  
  } catch (err) {  
    console.error(err);  
    showEmpty("Unable to load playlists", err.message);  
    setStatus("Connection error.");  
  } finally {  
    refresh.disabled = false;  
  }  
}

async function fetchPlaylistVideos(playlistId) {  
  const apiKey = getApiKey();  
  if (!apiKey) throw new Error("YouTube API key not found in AndroidBridge");  
  
  let allItems = [];  
  let pageToken = "";  
  
  do {  
    let url = "https://www.googleapis.com/youtube/v3/playlistItems?part=snippet,contentDetails&maxResults=50&playlistId=" + encodeURIComponent(playlistId) + "&key=" + encodeURIComponent(apiKey);  
    if (pageToken) url += "&pageToken=" + encodeURIComponent(pageToken);  
    const data = await apiRequest(url);  
    allItems = allItems.concat(data.items || []);  
    pageToken = data.nextPageToken || "";  
  } while (pageToken);  
  
  const videos = allItems  
    .map(item => {  
      const snippet = item.snippet || {};  
      const videoId = snippet.resourceId?.videoId || item.contentDetails?.videoId || "";  
      if (!videoId) return null;  
      const thumbnails = snippet.thumbnails || {};  
      return {  
        videoId,  
        title: snippet.title || "Untitled video",  
        thumbnail:  
          thumbnails.maxres?.url ||  
          thumbnails.standard?.url ||  
          thumbnails.high?.url ||  
          thumbnails.medium?.url ||  
          thumbnails.default?.url ||  
          `https://i.ytimg.com/vi/${encodeURIComponent(videoId)}/hqdefault.jpg`,  
        duration: "",  
        position: snippet.position ?? 0  
      };  
    })  
    .filter(Boolean);  
  
  for (let i = 0; i < videos.length; i += 50) {  
    const batch = videos.slice(i, i + 50);  
    const ids = batch.map(v => v.videoId).join(",");  
    const url = "https://www.googleapis.com/youtube/v3/videos?part=contentDetails&id=" + encodeURIComponent(ids) + "&key=" + encodeURIComponent(apiKey);  
    try {  
      const data = await apiRequest(url);  
      const durationMap = new Map((data.items || []).map(v => [v.id, v.contentDetails?.duration || ""]));  
      batch.forEach(video => {  
        video.duration = formatDuration(durationMap.get(video.videoId) || "");  
      });  
    } catch (err) {  
      console.warn("Durations unavailable", err);  
    }  
  }  
  return videos;  
}  
  
function formatDuration(iso) {  
  if (!iso) return "";  
  const match = iso.match(/PT(?:(\d+)H)?(?:(\d+)M)?(?:(\d+)S)?/);  
  if (!match) return "";  
  const hours = Number(match[1] || 0);  
  const minutes = Number(match[2] || 0);  
  const seconds = Number(match[3] || 0);  
  if (hours) return `${hours}:${String(minutes).padStart(2,"0")}:${String(seconds).padStart(2,"0")}`;  
  return `${minutes}:${String(seconds).padStart(2,"0")}`;  
}  
  
function renderVideos() {  
  currentView = "videos";  
  library.className = "grid";  
  library.innerHTML = "";  
  
  document.getElementById("itemCount").textContent = `${currentVideos.length} videos`;  
  
  currentVideos.forEach((video, index) => {  
    const card = document.createElement("button");  
    card.type = "button";  
    card.className = "card";  
    card.dataset.index = index;  
  
    const meta = video.duration ? `Video ${index + 1} · ${video.duration}` : `Video ${index + 1}`;  
  
    card.innerHTML = `  
      <div class="thumb">  
        <img loading="lazy" src="${safeText(video.thumbnail)}" alt="">  
        <span class="play-badge">  
          <svg viewBox="0 0 24 24" fill="currentColor"><path d="M7 4v16l13-8z"/></svg>  
        </span>  
      </div>  
      <div class="card-body">  
        <div class="card-title">${safeText(video.title)}</div>  
        <div class="card-meta">${safeText(meta)}</div>  
      </div>  
    `;  
    card.addEventListener("click", () => playSelectedVideo(index));  
    library.appendChild(card);  
  });  
  
  cards = [...library.querySelectorAll(".card")];  
  focusCard(0);  
}  
  
function focusCard(index) {  
  if (!cards.length) return;  
  selectedIndex = Math.max(0, Math.min(index, cards.length - 1));  
  cards.forEach((card, i) => card.classList.toggle("focused", i === selectedIndex));  
  cards[selectedIndex].focus({ preventScroll: true });  
  cards[selectedIndex].scrollIntoView({ block: "nearest", inline: "nearest" });  
}  
  
function loadYouTubeApi() {  
  if (window.YT && window.YT.Player) {  
    createOrStartPlayer();  
    return;  
  }  
  if (apiLoading) return;  
  apiLoading = true;  
  
  const script = document.createElement("script");  
  script.id = "yt-api";  
  script.src = "https://www.youtube.com/iframe_api";  
  window.onYouTubeIframeAPIReady = () => {  
    apiLoading = false;  
    createOrStartPlayer();  
  };  
  document.head.appendChild(script);  
}  
  
function resetUiTimer() {  
  playerScreen.classList.remove("hide-ui");  
  clearTimeout(uiHideTimer);  
  if (playerIsPlaying) {  
    uiHideTimer = setTimeout(() => {  
      playerScreen.classList.add("hide-ui");  
    }, 4500);  
  }  
}  
  
function createOrStartPlayer() {  
  if (!window.YT || !window.YT.Player) return;  
  
  if (!ytPlayer) {  
    ytPlayer = new YT.Player("youtubePlayer", {  
      width: "100%",  
      height: "100%",  
      host: "https://www.youtube-nocookie.com", // Suppresses tracking & targeted ads  
      playerVars: {  
        autoplay: 1,  
        controls: 1,  
        rel: 0,  
        playsinline: 1,  
        fs: 0, // Disabled: our custom container fills 100vw/100vh on TV  
        iv_load_policy: 3,  
        modestbranding: 1,  
        enablejsapi: 1  
      },  
      events: {  
        onReady: event => {  
          playerReady = true;  
          const iframe = event.target.getIframe();  
          if (iframe) {  
            iframe.style.width = "100%";  
            iframe.style.height = "100%";  
            iframe.setAttribute("allow", "autoplay; encrypted-media; picture-in-picture; fullscreen");  
          }  
          playVideoByIndex(pendingVideoIndex);  
        },  
        onStateChange: event => {  
          handlePlayerState(event.data);  
        },  
        onError: event => {  
          playerStatus.textContent = "Playback error (" + event.data + ").";  
        }  
      }  
    });  
  } else if (playerReady) {  
    playVideoByIndex(pendingVideoIndex);  
  }  
}  
  
function playSelectedVideo(index) {  
  if (!currentVideos.length || index < 0 || index >= currentVideos.length) return;  
  pendingVideoIndex = index;  
  currentVideoIndex = index;  
  
  playerScreen.classList.add("open");  
  document.body.style.overflow = "hidden";  
  
  updatePlayerInformation();  
  playerStatus.textContent = "Loading video…";  
  resetUiTimer();  
  loadYouTubeApi();  
}  
  
function playVideoByIndex(index) {  
  if (!playerReady || !currentVideos.length || index < 0 || index >= currentVideos.length) return;  
  currentVideoIndex = index;  
  const video = currentVideos[currentVideoIndex];  
  updatePlayerInformation();  
  
  playerStatus.textContent = `Video ${currentVideoIndex + 1} of ${currentVideos.length}`;  
  ytPlayer.loadVideoById({ videoId: video.videoId });  
  updatePlayerButtons();  
  resetUiTimer();  
}  
  
function updatePlayerInformation() {  
  const video = currentVideos[currentVideoIndex];  
  if (!video) return;  
  playingTitle.textContent = video.title;  
  playingSubtitle.textContent = `${currentPlaylist?.title || "Playlist"} · Video ${currentVideoIndex + 1} of ${currentVideos.length}`;  
  updatePlayerButtons();  
}  
  
function updatePlayerButtons() {  
  document.getElementById("previousVideo").disabled = currentVideoIndex <= 0;  
  document.getElementById("nextVideo").disabled = currentVideoIndex >= currentVideos.length - 1;  
  document.getElementById("togglePlayback").textContent = playerIsPlaying ? "Ⅱ Pause" : "▶ Play";  
}  
  
function handlePlayerState(state) {  
  if (!window.YT || !YT.PlayerState) return;  
  if (state === YT.PlayerState.PLAYING) {  
    playerIsPlaying = true;  
    playerStatus.textContent = `Playing ${currentVideoIndex + 1} of ${currentVideos.length}`;  
    updatePlayerButtons();  
    resetUiTimer();  
  } else if (state === YT.PlayerState.PAUSED) {  
    playerIsPlaying = false;  
    playerStatus.textContent = "Paused";  
    updatePlayerButtons();  
    resetUiTimer();  
  } else if (state === YT.PlayerState.ENDED) {  
    playerIsPlaying = false;  
    updatePlayerButtons();  
    resetUiTimer();  
    if (currentVideoIndex < currentVideos.length - 1) {  
      playVideoByIndex(currentVideoIndex + 1);  
    } else {  
      playerStatus.textContent = "Reached the end of this playlist.";  
    }  
  } else if (state === YT.PlayerState.BUFFERING) {  
    playerStatus.textContent = "Buffering…";  
  }  
}  
  
function playNextVideo() {  
  if (currentVideoIndex < currentVideos.length - 1) {  
    pendingVideoIndex = currentVideoIndex + 1;  
    playVideoByIndex(pendingVideoIndex);  
  }  
}  
  
function playPreviousVideo() {  
  if (currentVideoIndex > 0) {  
    pendingVideoIndex = currentVideoIndex - 1;  
    playVideoByIndex(pendingVideoIndex);  
  }  
}  
  
function togglePlayback() {  
  if (!ytPlayer || !playerReady) return;  
  resetUiTimer();  
  if (playerIsPlaying) {  
    ytPlayer.pauseVideo();  
  } else {  
    ytPlayer.playVideo();  
  }  
}  
  
function closePlayback() {  
  playerScreen.classList.remove("open");  
  playerScreen.classList.remove("hide-ui");  
  clearTimeout(uiHideTimer);  
  document.body.style.overflow = "";  
  
  if (ytPlayer && playerReady) {  
    ytPlayer.pauseVideo();  
  }  
  playerIsPlaying = false;  
  
  if (currentView === "videos") {  
    focusCard(currentVideoIndex);  
  }  
}  
  
function backToPlaylists() {  
  if (playerScreen.classList.contains("open")) {  
    closePlayback();  
  }  
  currentPlaylist = null;  
  currentVideos = [];  
  currentView = "playlists";  
  
  document.getElementById("hero").style.display = "";  
  document.getElementById("browseBar").classList.remove("visible");  
  document.getElementById("sectionTitle").textContent = "Explore playlists";  
  setStatus("Select a playlist to browse its videos.");  
  renderLibrary();  
}  
  
document.getElementById("refreshBtn").addEventListener("click", loadLibrary);  
document.getElementById("closePlayer").addEventListener("click", closePlayback);  
document.getElementById("backToPlaylists").addEventListener("click", backToPlaylists);  
document.getElementById("nextVideo").addEventListener("click", playNextVideo);  
document.getElementById("previousVideo").addEventListener("click", playPreviousVideo);  
document.getElementById("togglePlayback").addEventListener("click", togglePlayback);  
  
document.addEventListener("keydown", e => {  
  const playerOpen = playerScreen.classList.contains("open");  
  
  resetUiTimer();  
  
  if (["Escape", "Backspace", "BrowserBack", "GoBack"].includes(e.key)) {  
    e.preventDefault();  
    if (playerOpen) {  
      closePlayback();  
    } else if (currentView === "videos") {  
      backToPlaylists();  
    }  
    return;  
  }  
  
  if (playerOpen) {  
    if (e.key === "MediaPlayPause" || e.key === " ") {  
      e.preventDefault();  
      togglePlayback();  
    } else if (e.key === "MediaTrackNext" || e.key === "ArrowRight") {  
      if (document.activeElement?.tagName !== "BUTTON") {  
        e.preventDefault();  
        playNextVideo();  
      }  
    } else if (e.key === "MediaTrackPrevious" || e.key === "ArrowLeft") {  
      if (document.activeElement?.tagName !== "BUTTON") {  
        e.preventDefault();  
        playPreviousVideo();  
      }  
    }  
    return;  
  }  
  
  if (!cards.length) {  
    if (e.key === "Enter") {  
      e.preventDefault();  
      if (currentView === "videos" && currentPlaylist) {  
        openPlaylist(currentPlaylist);  
      } else {  
        loadLibrary();  
      }  
    }  
    return;  
  }  
  
  const cols = getComputedStyle(library).gridTemplateColumns.split(" ").length || 1;  
  
  if (e.key === "ArrowRight") {  
    e.preventDefault();  
    focusCard(selectedIndex + 1);  
  } else if (e.key === "ArrowLeft") {  
    e.preventDefault();  
    focusCard(selectedIndex - 1);  
  } else if (e.key === "ArrowDown") {  
    e.preventDefault();  
    focusCard(selectedIndex + cols);  
  } else if (e.key === "ArrowUp") {  
    e.preventDefault();  
    focusCard(selectedIndex - cols);  
  } else if (e.key === "Enter" || e.key === " ") {  
    e.preventDefault();  
    cards[selectedIndex].click();  
  }  
});  
  
window.addEventListener("load", loadLibrary);  
})();
